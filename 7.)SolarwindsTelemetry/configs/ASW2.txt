! ==============================================================================
! Device: ASW2 (Middle-Left Access Switch - Dual-Homed Workload Tier)
! Lab 07: Enterprise Telemetry, SNMP Traps & Informs to SolarWinds Orion
! ==============================================================================
hostname ASW2

no ip routing

! --- VLAN Database ---
vlan 10
 name NMS_Management
 exit
vlan 20
 name Linux_Workloads
 exit
vlan 99
 name Native_Trunk_Control
 exit

! --- Management SVI ---
interface Vlan10
 ip address 10.0.10.12 255.255.255.0
 no shutdown
 exit

ip default-gateway 10.0.10.1

! --- Redundant Trunks to DSW1 and DSW2 ---
interface GigabitEthernet0/0
 description Primary Trunk Uplink to DSW1 (Gi0/3)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 switchport trunk native vlan 99
 no shutdown
 exit

interface GigabitEthernet0/1
 description Secondary Trunk Uplink to DSW2 (Gi1/1)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 switchport trunk native vlan 99
 no shutdown
 exit

! --- Access Port for Linux-01 (Gi0/2) ---
interface GigabitEthernet0/2
 description Linux-01 Workload Host (e0)
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
 exit

! --- SNMP Polling & Traps ---
snmp-server community SolarReadOnly RO
snmp-server location Access Tier - Workload Rack 1
snmp-server contact Philippe Truong (NetOps)

snmp-server enable traps snmp authentication linkdown linkup
snmp-server trap-source Vlan10
snmp-server host 10.0.10.50 version 2c SolarReadOnly
snmp-server host 10.0.10.50 informs version 2c SolarReadOnly

logging host 10.0.10.50
